% --- Sprache und Schriftart ---
\usepackage{newtxtext} % Times New Roman Alternative
\usepackage{newtxmath} % Passendes Mathe zur Schriftart
\usepackage[ngerman]{babel} % Deutsche Sprache und Silbentrennung
\usepackage[T1]{fontenc} % Umlaute korrekt

% --- Überschriften ---
\setkomafont{disposition}{\rmfamily\bfseries}
\setkomafont{chapter}{\fontsize{14pt}{17pt}\selectfont}
\setkomafont{section}{\fontsize{14pt}{17pt}\selectfont}
\setkomafont{subsection}{\fontsize{12pt}{15pt}\selectfont}
\setkomafont{subsubsection}{\fontsize{12pt}{15pt}\selectfont}
\RedeclareSectionCommand[
  beforeskip=0pt % Zieht die Überschrift ganz nach oben an den Rand
]{chapter}

% --- Seitenformat und Abstände ---
% Provadis Vorgabe: Links/Oben 3cm, Rechts/Unten 2cm
\usepackage[left=3cm,right=2cm,top=3cm,bottom=2cm]{geometry}

\usepackage{setspace}
\onehalfspacing
\flushbottom

% --- Grafiken und Tabellen ---
\usepackage{graphicx}
\usepackage{float}
\usepackage{longtable}
\usepackage{booktabs}
\usepackage{subcaption}
\usepackage[format=plain,labelfont=bf]{caption}
\captionsetup[table]{position=above}

% --- Mathematik ---
\usepackage{amsmath}
\usepackage{icomma}
\usepackage{siunitx}
\sisetup{locale = DE}

% --- Verzeichnisse und Zitate ---
\usepackage[
  style=authoryear,
  backend=biber,
  maxbibnames=99,
  maxcitenames=2,
  autocite=footnote,
  giveninits=true,
  urldate=short]{biblatex}
\usepackage{csquotes}
\addbibresource{Referenzen.bib}
\renewbibmacro*{cite:labeldate+extradate}{%
  \iffieldundef{labelyear}
    {}
    {\printtext[bibhyperref]{%
       \mkbibparens{%
         \printfield{labelyear}%
         \printfield{extradate}}}}}
\DeclareFieldFormat{urldate}{\mkbibbrackets{\bibstring{urlseen}\space#1}}
\setlength{\bibitemsep}{0.7em}
\setcounter{biburllcpenalty}{7000}

\DefineBibliographyStrings{ngerman}{
  andothers = {{et\,al\adddot}},
}
\usepackage[acronym, toc]{glossaries}
\renewcommand*{\glspostdescription}{\dotfill}

% --- Quellcode ---
\usepackage{listings}
\usepackage{xcolor}
\definecolor{codegreen}{rgb}{0,0.6,0}
\definecolor{codegray}{rgb}{0.5,0.5,0.5}
\definecolor{codepurple}{rgb}{0.58,0,0.82}
\definecolor{backcolour}{rgb}{0.95,0.95,0.92}
\lstdefinestyle{myStyle}{
    backgroundcolor=\color{backcolour},
    commentstyle=\color{codegreen},
    keywordstyle=\color{magenta},
    numberstyle=\tiny\color{codegray},
    stringstyle=\color{codepurple},
    basicstyle=\ttfamily\footnotesize,
    breakatwhitespace=false,
    breaklines=true,
    keepspaces=true,
    numbers=left,
    numbersep=5pt,
    showspaces=false,
    showstringspaces=false,
    showtabs=false,
    tabsize=2,
}
\lstset{style=myStyle}

% --- Zeug ---
\usepackage{titling}
\usepackage{pdfpages}
\setlength{\marginparwidth}{2cm}
\usepackage{todonotes} % Randnotizen während des Schreibens
\setuptodonotes{fancyline}
